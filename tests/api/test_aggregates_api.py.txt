"""Test Aggregates (Custom Bars / OHLCV) API."""

from datetime import UTC, date, datetime
from typing import Any

import pytest
from conftest import MockApiFactory
from pydantic import ValidationError

from massive_api.api.aggregates import AggregatesApi, Bar

SAMPLE_BAR = {
    "o": 226.0,
    "h": 230.5,
    "l": 225.1,
    "c": 229.8,
    "v": 44304723,
    "vw": 228.53,
    "n": 512345,
    "t": 1735689600000,  # Unix ms for 2025-01-01T00:00:00Z
    "otc": False,
}

# Required path arguments shared across cases; query params vary per test.
BASE_KWARGS = {
    "ticker": "AAPL",
    "multiplier": 1,
    "timespan": "day",
    "from_date": "2025-01-01",
    "to_date": "2025-01-31",
}
BASE_ENDPOINT = "v2/aggs/ticker/AAPL/range/1/day/2025-01-01/2025-01-31"


@pytest.mark.asyncio
@pytest.mark.parametrize(
    ("kwargs", "expected_params"),
    [
        # Every call sends the page-size `limit`, defaulting to the API maximum (50000).
        ({}, {"limit": "50000"}),
        # `adjusted` serializes to a "true"/"false" string.
        ({"adjusted": True}, {"adjusted": "true", "limit": "50000"}),
        ({"adjusted": False}, {"adjusted": "false", "limit": "50000"}),
        # `order` is a plain direction that maps onto the endpoint's `sort` param.
        ({"order": "desc"}, {"sort": "desc", "limit": "50000"}),
        ({"order": "asc"}, {"sort": "asc", "limit": "50000"}),
        # `max_results` caps client-side only; the wire `limit` always stays at the API max,
        # since a small `limit` bounds base aggregates (baked into the cursor) and can starve
        # multiplier>1 aggregation. The cap is still threaded to `_get_all_pages` (asserted below).
        ({"max_results": 25}, {"limit": "50000"}),
    ],
)
async def test_custom_bars_parameters(
    mock_api_factory: MockApiFactory,
    kwargs: dict[str, Any],
    expected_params: dict[str, str],
) -> None:
    """get_custom_bars builds the correct endpoint, params, and results cap."""
    api, mocks = mock_api_factory.create(AggregatesApi, mock_pages=[SAMPLE_BAR])

    result = await api.get_custom_bars(**BASE_KWARGS, **kwargs)

    mocks.get_all_pages.assert_called_once_with(
        BASE_ENDPOINT,
        expected_params,
        max_results=kwargs.get("max_results"),
    )
    assert len(result) == 1
    assert isinstance(result[0], Bar)


@pytest.mark.asyncio
@pytest.mark.parametrize(
    ("path_kwargs", "expected_endpoint"),
    [
        # multiplier + timespan are interpolated into the range path segment.
        ({"multiplier": 5, "timespan": "minute"}, "v2/aggs/ticker/AAPL/range/5/minute/2025-01-01/2025-01-31"),
        ({"ticker": "MSFT", "timespan": "week"}, "v2/aggs/ticker/MSFT/range/1/week/2025-01-01/2025-01-31"),
        # date/datetime objects for from/to coerce to ISO "YYYY-MM-DD".
        ({"from_date": date(2025, 1, 1), "to_date": date(2025, 1, 31)}, BASE_ENDPOINT),
    ],
)
async def test_custom_bars_builds_endpoint_path(
    mock_api_factory: MockApiFactory,
    path_kwargs: dict[str, Any],
    expected_endpoint: str,
) -> None:
    """The ticker/multiplier/timespan/from/to path parameters are encoded into the URL."""
    api, mocks = mock_api_factory.create(AggregatesApi, mock_pages=[SAMPLE_BAR])

    await api.get_custom_bars(**{**BASE_KWARGS, **path_kwargs})

    mocks.get_all_pages.assert_called_once_with(
        expected_endpoint,
        {"limit": "50000"},
        max_results=None,
    )


@pytest.mark.asyncio
async def test_custom_bars_raw_returns_untouched_records(mock_api_factory: MockApiFactory) -> None:
    """get_custom_bars_raw returns the untouched page dicts (no model conversion)."""
    api, mocks = mock_api_factory.create(AggregatesApi, mock_pages=[SAMPLE_BAR])

    result = await api.get_custom_bars_raw(**BASE_KWARGS)

    mocks.get_all_pages.assert_called_once_with(
        BASE_ENDPOINT,
        {"limit": "50000"},
        max_results=None,
    )
    assert result == [SAMPLE_BAR]


@pytest.mark.asyncio
async def test_custom_bars_parses_typed_fields(mock_api_factory: MockApiFactory) -> None:
    """A valid record parses single-letter keys into readable, precisely typed fields."""
    api, _ = mock_api_factory.create(AggregatesApi, mock_pages=[SAMPLE_BAR])

    bar = (await api.get_custom_bars(**BASE_KWARGS))[0]

    assert bar.open == 226.0
    assert bar.high == 230.5
    assert bar.low == 225.1
    assert bar.close == 229.8
    assert bar.volume == 44304723
    assert bar.vwap == 228.53
    assert bar.transactions == 512345
    # The Unix ms timestamp is parsed into a timezone-aware UTC datetime.
    assert bar.timestamp == datetime(2025, 1, 1, tzinfo=UTC)
    assert bar.otc is False


@pytest.mark.asyncio
async def test_custom_bars_tolerates_missing_optional_fields(mock_api_factory: MockApiFactory) -> None:
    """vwap/transactions/otc may be absent; the OHLCV core still validates."""
    core_only = {"o": 226.0, "h": 230.5, "l": 225.1, "c": 229.8, "v": 44304723, "t": 1735707600000}
    api, _ = mock_api_factory.create(AggregatesApi, mock_pages=[core_only])

    bar = (await api.get_custom_bars(**BASE_KWARGS, on_validation_error="raise"))[0]

    assert bar.vwap is None
    assert bar.transactions is None
    assert bar.otc is None


@pytest.mark.parametrize(
    "bad_record",
    [
        {**SAMPLE_BAR, "c": "expensive"},  # not a number
        {**SAMPLE_BAR, "t": "not-a-timestamp"},  # not an integer timestamp
        {k: v for k, v in SAMPLE_BAR.items() if k != "o"},  # missing required open
    ],
)
@pytest.mark.asyncio
async def test_custom_bars_rejects_invalid_records(
    mock_api_factory: MockApiFactory,
    bad_record: dict[str, Any],
) -> None:
    """The hardened Bar model raises on an invalid/missing core field under 'raise' mode."""
    api, _ = mock_api_factory.create(AggregatesApi, mock_pages=[bad_record])

    with pytest.raises(ValidationError):
        await api.get_custom_bars(**BASE_KWARGS, on_validation_error="raise")


@pytest.mark.parametrize(
    "bad_params",
    [
        {"order": "dec"},  # not an Order ("asc"/"desc")
        {"timespan": "fortnight"},  # not a Timespan
        {"multiplier": 0},  # must be a positive integer
        {"multiplier": -5},  # must be a positive integer
        {"from_date": "not-a-date"},  # unparseable date
        {"to_date": "2025-13-01"},  # out-of-range month
        {"max_results": 0},  # must be a positive integer
        {"max_results": -5},  # must be a positive integer
    ],
)
@pytest.mark.asyncio
async def test_custom_bars_rejects_invalid_parameters(
    mock_api_factory: MockApiFactory,
    bad_params: dict[str, Any],
) -> None:
    """An invalid path or query parameter raises before any request is made."""
    api, _ = mock_api_factory.create(AggregatesApi)

    with pytest.raises(ValueError):  # noqa: PT011
        await api.get_custom_bars(**{**BASE_KWARGS, **bad_params}, on_validation_error="raise")
